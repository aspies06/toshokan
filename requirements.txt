# Python dependencies for the ingestion pipeline.
#
# The pipeline owns document ingestion end to end: parsing, chunking, embedding
# and writing to SQLite. The Electron main process only reads from the database.
#
# torch is not listed explicitly but is installed transitively twice over:
# docling depends on docling-ibm-models for PDF layout analysis, and
# sentence-transformers needs it for the embedding model. Expect a multi-GB
# install.
#
# Install the production environment with:
#   python -m pip install -r requirements.txt
# Or install the development environment with:
#   python -m pip install -r requirements-dev.txt

# Document conversion (PDF, DOCX, HTML, ...) into a DoclingDocument.
docling>=2.118,<3

# HybridChunker plus the tokenizer used to size chunks.
docling-core[chunking]>=2.91,<3

# Embedding model. Vectors are L2-normalized so that the squared-L2 distance
# sqlite-vec computes ranks identically to cosine similarity.
sentence-transformers>=3.0,<6

# Loadable SQLite extension providing the vec0 virtual table. Kept in step with
# the sqlite-vec version in package.json, since both processes open the same
# database file.
sqlite-vec>=0.1.9,<0.2

# Verifies TLS against the OS trust store instead of Python's bundled one, so
# model downloads work behind the TLS-inspecting proxies common on corporate
# networks. Optional at runtime; see src/pipeline/runtime.py.
truststore>=0.10,<0.11
